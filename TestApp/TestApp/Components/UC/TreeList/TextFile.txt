<div class="treelist-level">
    @if (Items != null)
    {
        foreach (var item in Items)
        {
            var isSelected = IsSelected(item);
            <!--trên desktop-->

            <div class="treelist-row @(isSelected ? "selected" : "") d-none d-md-grid"
                 style="grid-template-columns:@GetGridTemplate(); padding-left:@(Level* IndentPx)px;"
                 @onclick="() => OnTouch(item)">

                @RenderExpander(item)

                @for (int i = 0; i < _normalProps.Count; i++)
                {
                    @RenderCell(item, _normalProps[i], i, isMobile: false)
                }

                @RenderButtons(item)
            </div>


            <!--trên mobile-->
            <div class="treelist-row @(isSelected ? "selected" : "") d-flex d-md-none"
                 style="padding-left:@(Level* IndentPx)px;"
                 @onclick="() => OnTouch(item)">

                @RenderExpander(item)

                <div class="d-block w-100">
                    @for (int i = 0; i < _normalProps.Count; i++)
                    {
                        @RenderCell(item, _normalProps[i], i, isMobile: true)
                    }

                    @RenderButtons(item, isMobile: true)
                </div>
            </div>

            @if (HasChild(item) && IsExpanded(item))
            {
                var propName = ChildPropertyNames![Level];
                var prop = item.GetType().GetProperty(propName);

                if (prop?.GetValue(item) is IEnumerable seq && seq.Cast<object>().Any())
                {
                    var childType = GetElementType(prop.PropertyType);
                    @DynamicChild(childType!, seq, Level + 1)
                }
            }
        }
    }
</div>

@code {
    [Parameter] public List<List<string>>? DisplayProperties { get; set; }
    [Parameter] public IReadOnlyList<TItem>? Items { get; set; }
    [Parameter] public int Level { get; set; }
    [Parameter] public int IndentPx { get; set; }
    [Parameter] public int ExpanderWidth { get; set; }
    [Parameter] public List<List<int>>? ColumnWidths { get; set; }
    [Parameter] public string? SelectedRowBg { get; set; }
    [Parameter] public List<string> TextCSS { get; set; } = [];

    [Parameter] public List<bool> ShowEdit { get; set; } = [];
    [Parameter] public List<bool> ShowDelete { get; set; } = [];

    [Parameter] public List<string>? Headers { get; set; }
    [Parameter] public List<string>? ChildPropertyNames { get; set; }
    // Khởi tạo mặc định để tránh null reference nếu parent quên truyền
    [Parameter] public TreeSelectionState SelectionState { get; set; } = new TreeSelectionState();
    [Parameter] public List<string>? KeyFieldNames { get; set; }

    [Parameter] public EventCallback<object> OnSelectionChanged { get; set; }
    [Parameter] public EventCallback<object> OnRowClicked { get; set; }
    [Parameter] public EventCallback<object> OnRowDbleClicked { get; set; }
    [Parameter] public EventCallback<object> OnEditClicked { get; set; }
    [Parameter] public EventCallback<object> OnDeleteClicked { get; set; }

    private readonly HashSet<object> _expanded = new();
    private List<PropertyInfo> _normalProps = new();
    private List<PropertyInfo> _childProps = new();

    // để quản lý subscribe/unsubscribe chính xác
    private Action? _selectionUpdatedHandler;
    private TreeSelectionState? _prevSelectionState;

    RenderFragment RenderExpander(object item) => builder =>
    {
        builder.OpenElement(0, "div");
        builder.AddAttribute(1, "class", "tl-cell tl-expander");

        if (HasChild(item))
        {
            builder.OpenElement(2, "i");
            builder.AddAttribute(3, "class",
                $"fa {(IsExpanded(item) ? "fa-minus-square" : "fa-plus-square")} tl-expander-icon");
            builder.AddAttribute(4, "onclick",
                EventCallback.Factory.Create(this, () => ToggleExpand(item)));
            builder.AddEventStopPropagationAttribute(5, "onclick", true);
            builder.CloseElement();
        }

        builder.CloseElement();
    };

    RenderFragment RenderCell(object item, PropertyInfo prop, int index, bool isMobile) => builder =>
    {
        var value = prop.GetValue(item);
        var (text, css) = FormatCell(value, index, isMobile);

        builder.OpenElement(0, "div");
        builder.AddAttribute(1, "class", $"tl-cell {css}");
        builder.AddContent(2, text);
        builder.CloseElement();
    };
    private (string Text, string Css) FormatCell(object? value, int index, bool isMobile)
    {
        string css = "";
        if (TextCSS.Count>0 && TextCSS.Count > Level && !string.IsNullOrEmpty(TextCSS[Level]))
            css = TextCSS[Level];

        if (value == null)
            return ("", "");

        // NUMBER
        if (value is int or long or short)
            return ($"{value:N0}", $"text-end {css}");

        if (value is decimal d)
            return ($"{d:N0}", $"text-end {css}");

        if (value is double db)
            return ($"{db:N0}", $"text-end {css}");

        if (value is float f)
            return ($"{f:N0}", $"text-end {css}");

        // DATE
        if (value is DateTime dt)
            return (dt.ToString("dd/MM/yyyy"), css);

        // STRING
        if (Level == 0)
            return (value.ToString()!, $"top-level {css}");

        if (isMobile && index == 0)
            return (value.ToString()!, css);

        return (value.ToString()!, css);
    }
    RenderFragment RenderButtons(TItem item, bool isMobile = false) => builder =>
    {
        if (ShowEdit[Level])
        {
            builder.OpenElement(0, "button");
            builder.AddAttribute(1, "class",
                "btn btn-primary p-0 d-flex align-items-center justify-content-center ms-2");
            builder.AddAttribute(2, "style", "width:32px;height:32px;");
            builder.AddAttribute(3, "onclick",
                EventCallback.Factory.Create(this, () => OnEditClick(item)));

            if (isMobile)
                builder.AddEventStopPropagationAttribute(4, "onclick", true);

            builder.OpenElement(5, "i");
            builder.AddAttribute(6, "class", "bi bi-pencil");
            builder.CloseElement();

            builder.CloseElement();
        }


        if (ShowDelete[Level])
        {
            builder.OpenElement(10, "button");
            builder.AddAttribute(11, "class",
                "btn btn-danger p-0 d-flex align-items-center justify-content-center ms-2");
            builder.AddAttribute(12, "style", "width:32px;height:32px;");
            builder.AddAttribute(13, "onclick",
                EventCallback.Factory.Create(this, () => OnDeleteClick(item)));

            if (isMobile)
                builder.AddEventStopPropagationAttribute(14, "onclick", true);

            builder.OpenElement(15, "i");
            builder.AddAttribute(16, "class", "bi bi-trash3-fill");
            builder.CloseElement();

            builder.CloseElement();
        }
    };


    protected override void OnParametersSet()
    {
        SelectionState ??= new TreeSelectionState();

        if (!ReferenceEquals(_prevSelectionState, SelectionState))
        {
            if (_prevSelectionState != null && _selectionUpdatedHandler != null)
                _prevSelectionState.SelectionUpdated -= _selectionUpdatedHandler;

            _selectionUpdatedHandler ??= () => InvokeAsync(StateHasChanged);
            SelectionState.SelectionUpdated += _selectionUpdatedHandler;
            _prevSelectionState = SelectionState;
        }

        var allProps = typeof(TItem).GetProperties(BindingFlags.Public | BindingFlags.Instance);

        // Xác định cột hiển thị (normal props)
        if (DisplayProperties != null && Level < DisplayProperties.Count)
        {
            var levelProps = DisplayProperties[Level];
            _normalProps = levelProps
                .Select(name => allProps.FirstOrDefault(p => p.Name == name))
                .Where(p => p != null && (!typeof(IEnumerable).IsAssignableFrom(p.PropertyType) || p.PropertyType == typeof(string)))
                .ToList()!;
        }
        else
        {
            _normalProps = allProps
                .Where(p => !typeof(IEnumerable).IsAssignableFrom(p.PropertyType) || p.PropertyType == typeof(string))
                .ToList();
        }

        // child props: tất cả collection props trừ TypeList
        _childProps = allProps
            .Where(p => typeof(IEnumerable).IsAssignableFrom(p.PropertyType)
                     && p.PropertyType != typeof(string))
            .ToList();


        SelectionState.OnSelectionChanged = OnSelectionChanged;
        if (_expanded.Count == 0 && Items != null)
        {
            ExpandAll(Items);
        }
        if (Items != null)
        {
            foreach (var item in Items.Cast<object>())
            {
                if (HasChild(item) && !_expanded.Contains(item))
                {
                    _expanded.Add(item);
                }
            }
        }
    }

    private void ExpandAll(IEnumerable items)
    {
        foreach (var item in items.Cast<object>())
        {
            if (HasChild(item))
            {
                _expanded.Add(item);

                var props = item.GetType()
                                .GetProperties(BindingFlags.Public | BindingFlags.Instance)
                                .Where(p => typeof(IEnumerable).IsAssignableFrom(p.PropertyType)
                                         && p.PropertyType != typeof(string));

                foreach (var childProp in props)
                {
                    if (childProp.GetValue(item) is IEnumerable seq && seq.Cast<object>().Any())
                    {
                        ExpandAll(seq);
                    }
                }
            }
        }
    }

    private async Task OnRowDbleClick(TItem item)
    {
        await SelectionState.SelectAsync(item);
        await OnRowDbleClicked.InvokeAsync(item);
        await InvokeAsync(StateHasChanged);
    }
    //xử lý double tap trên di động
    #region Xử lý double tap trên di động
    private DateTime _lastTap = DateTime.MinValue;
    private const int TapDelay = 300; // ms
    private async Task OnTouch(TItem item)
    {
        await SelectionState.SelectAsync(item);
        await OnRowClicked.InvokeAsync(item);
        var now = DateTime.Now;
        if ((now - _lastTap).TotalMilliseconds <= TapDelay)
        {
            // Đây là double tap trên mobile
            await OnRowDbleClick(item);
        }
        _lastTap = now;
        await InvokeAsync(StateHasChanged);
    }
    #endregion
    private async Task OnEditClick(TItem item)
    {
        await OnEditClicked.InvokeAsync(item);
        await InvokeAsync(StateHasChanged);
    }
    private async Task OnDeleteClick(TItem item)
    {
        await OnDeleteClicked.InvokeAsync(item);
        await InvokeAsync(StateHasChanged);
    }

    private void ToggleExpand(object item)
    {
        if (!_expanded.Add(item))
            _expanded.Remove(item);
    }

    private bool IsExpanded(object item) => _expanded.Contains(item);

    private bool HasChild(object item)
    {
        if (ChildPropertyNames == null || Level >= ChildPropertyNames.Count)
            return false;

        var propName = ChildPropertyNames[Level];
        var prop = item.GetType().GetProperty(propName);

        if (prop == null)
            return false;

        var val = prop.GetValue(item) as IEnumerable;
        return val != null && val.Cast<object>().Any();
    }


    private IEnumerable<PropertyInfo> GetNormalProperties() =>
        typeof(TItem).GetProperties()
            .Where(p => !typeof(IEnumerable).IsAssignableFrom(p.PropertyType) || p.PropertyType == typeof(string));

    private IEnumerable<PropertyInfo> GetChildProperties() =>
        typeof(TItem).GetProperties()
            .Where(p => typeof(IEnumerable).IsAssignableFrom(p.PropertyType) && p.PropertyType != typeof(string));

    private static Type? GetElementType(Type t)
    {
        if (t.IsArray)
            return t.GetElementType();
        if (t.IsGenericType)
            return t.GetGenericArguments().FirstOrDefault();
        return typeof(object);
    }

    private static string Format(object? v) =>
        v switch
        {
            null => "",
            DateTime dt => dt.ToString("dd/MM/yyyy"),
            bool b => b ? "True" : "False",
            _ => v?.ToString() ?? ""
        };

    private string GetGridTemplate()
    {
        var cols = new List<string>();

        // 1. Expander
        var expander = ExpanderWidth > 30 ? ExpanderWidth : 30;
        cols.Add($"{expander}px");

        // 2. Columns theo tỷ lệ
        if (ColumnWidths != null && Level < ColumnWidths.Count)
        {
            var widths = ColumnWidths[Level]
                         .Take(_normalProps.Count)
                         .ToList();

            foreach (var w in widths)
            {
                cols.Add($"{w}fr");
            }
        }
        else
        {
            // auto chia đều
            int count = _normalProps.Count;
            for (int i = 0; i < count; i++)
                cols.Add("1fr");
        }

        // 3. Buttons
        if (ShowDelete[Level])
            cols.Add("40px");
        if (ShowEdit[Level])
            cols.Add("40px");

        return string.Join(" ", cols);
    }

    private bool IsSelected(object item)
    {
        return ReferenceEquals(SelectionState.SelectedItem, item);
    }

    RenderFragment DynamicChild(Type childType, IEnumerable data, int childLevel) => builder =>
    {
        var seq = 0;
        var generic = typeof(TreeListView<>).MakeGenericType(childType);
        builder.OpenComponent(seq++, generic);

        builder.AddAttribute(seq++, "Items", data);
        builder.AddAttribute(seq++, "Level", childLevel);
        builder.AddAttribute(seq++, "IndentPx", IndentPx);
        builder.AddAttribute(seq++, "ColumnWidths", ColumnWidths);
        builder.AddAttribute(seq++, "ExpanderWidth", ExpanderWidth);
        builder.AddAttribute(seq++, "SelectedRowBg", SelectedRowBg);
        builder.AddAttribute(seq++, "SelectionState", SelectionState);
        builder.AddAttribute(seq++, "KeyFieldNames", KeyFieldNames);
        builder.AddAttribute(seq++, "OnSelectionChanged", OnSelectionChanged);
        builder.AddAttribute(seq++, "OnRowClicked", OnRowClicked);
        builder.AddAttribute(seq++, "OnRowDbleClicked", OnRowDbleClicked);
        builder.AddAttribute(seq++, "DisplayProperties", DisplayProperties);
        builder.AddAttribute(seq++, "Headers", Headers);
        builder.AddAttribute(seq++, "ShowDelete", ShowDelete);
        builder.AddAttribute(seq++, "ShowEdit", ShowEdit);
        builder.AddAttribute(seq++, "OnEditClicked", OnEditClicked);
        builder.AddAttribute(seq++, "OnDeleteClicked", OnDeleteClicked);
        builder.AddAttribute(seq++, "ChildPropertyNames", ChildPropertyNames);
        builder.AddAttribute(seq++, "TextCSS", TextCSS);

        builder.CloseComponent();
    };


    public void Dispose()
    {
        if (_prevSelectionState != null && _selectionUpdatedHandler != null)
        {
            _prevSelectionState.SelectionUpdated -= _selectionUpdatedHandler;
        }
    }
}
